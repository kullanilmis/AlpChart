// Hacim profili kutularını temizle
clearVolumeProfile() =>
    if array.size(vpBoxes) > 0

            box.delete(array.get(vpBoxes, i))
        array.clear(vpBoxes)

// Geçmiş kutu ekleme fonksiyonu
addToHistory(bool isLong) =>
    if showHistoryBoxes and not na(currentBox)
        // Mevcut kutunun bilgilerini al
        boxTop = box.get_top(currentBox)
        boxBottom = box.get_bottom(currentBox)
        boxLeft = box.get_left(currentBox)
        boxRight = box.get_right(currentBox)
        
        // Eski kutuyu sil
        box.delete(currentBox)
        
        // Geçmiş renkleri belirle
        histBgColor = isLong ? histLongBoxColor : histShortBoxColor
        histBorderColor = isLong ? histLongBorder : histShortBorder
        
        // Yeni geçmiş kutusu oluştur (saydam renklerle)
        histBox = box.new(boxLeft, boxTop, boxRight, boxBottom,
             border_color = histBorderColor,
             bgcolor = histBgColor,
             border_width = 1,
             border_style = line.style_solid)
        
        array.push(historyBoxes, histBox)
        
        // Seviye çizgilerini geçmişe ekle (renkleri değiştir)
        if not na(level25Line)
            line.set_color(level25Line, histLevel25)
            array.push(historyLevel25, level25Line)
        else
            array.push(historyLevel25, line(na))
            
        if not na(level50Line)
            line.set_color(level50Line, histLevel50)
            array.push(historyLevel50, level50Line)
        else
            array.push(historyLevel50, line(na))
            
        if not na(level75Line)
            line.set_color(level75Line, histLevel75)
            array.push(historyLevel75, level75Line)
        else
            array.push(historyLevel75, line(na))
        
        // Geçmiş kutu sayısı limiti aşarsa en eskisini sil
        if array.size(historyBoxes) > historyBoxCount
            oldBox = array.shift(historyBoxes)
            oldL25 = array.shift(historyLevel25)
            oldL50 = array.shift(historyLevel50)
            oldL75 = array.shift(historyLevel75)
            
            box.delete(oldBox)
            line.delete(oldL25)
            line.delete(oldL50)
            line.delete(oldL75)

// Kutu çizim fonksiyonu
drawZigzagBox(float topPrice, float bottomPrice, int leftBar, int rightBar, bool isLong) =>
    box resultBox = na
    line resultL25 = na
    line resultL50 = na
    line resultL75 = na
    
    if showBox
        // Mevcut kutuyu geçmişe ekle (eğer varsa)
        if not na(currentBox)
            addToHistory(currentBoxIsLong)
        
        // Eğer geçmiş kutular gösterilmeyecekse eski kutuyu sil
        if not showHistoryBoxes
            if not na(currentBox)
                box.delete(currentBox)
            if not na(level25Line)
                line.delete(level25Line)
            if not na(level50Line)
                line.delete(level50Line)
            if not na(level75Line)
                line.delete(level75Line)
        
        // Kutu renklerini belirle
        bgColor = isLong ? longBoxColor : shortBoxColor
        borderColor = isLong ? longBoxBorder : shortBoxBorder
        
        // Yeni kutu çiz
        resultBox := box.new(leftBar, topPrice, rightBar, bottomPrice, 
             border_color = borderColor, 
             bgcolor = bgColor, 
             border_width = 1,
             border_style = line.style_solid,
             extend = extend.none)
        
        // Fibonacci seviyelerini hesapla
        priceRange = topPrice - bottomPrice
        l25 = bottomPrice + priceRange * 0.25
        l50 = bottomPrice + priceRange * 0.50
        l75 = bottomPrice + priceRange * 0.75
        
        // Seviye çizgilerini oluştur
        if showLevels
            resultL25 := line.new(leftBar, l25, rightBar, l25, 
                 color = level25Color, 
                 style = levelStyle, 
                 width = 1)
            resultL50 := line.new(leftBar, l50, rightBar, l50, 
                 color = level50Color, 
                 style = levelStyle, 
                 width = 2)
            resultL75 := line.new(leftBar, l75, rightBar, l75, 
                 color = level75Color, 
                 style = levelStyle, 
                 width = 1)
    
    [resultBox, resultL25, resultL50, resultL75]

// Hacim profili çizim fonksiyonu
drawVolumeProfile(int startBar, int endBar, float highPrice, float lowPrice) =>
    line resultPocLine = na
    line resultVahLine = na
    line resultValLine = na
    
    if showVolumeProfile and startBar < endBar and highPrice > lowPrice
        // Önceki profili temizle
        clearVolumeProfile()
        
        if not na(pocLine)
            line.delete(pocLine)
        if not na(vahLine)
            line.delete(vahLine)
        if not na(valLine)
            line.delete(valLine)
        
        // Fiyat aralığını satırlara böl
        priceRange = highPrice - lowPrice
        rowHeight = priceRange / vpRowCount
        
        // Her satır için hacim array'leri
        upVolumes = array.new_float(vpRowCount, 0.0)
        downVolumes = array.new_float(vpRowCount, 0.0)
        totalVolumes = array.new_float(vpRowCount, 0.0)
        
        // Bar sayısını hesapla
        barCount = endBar - startBar
        
        // Her bar için hacmi ilgili fiyat seviyesine dağıt
        for i = 0 to barCount
            offset = endBar - startBar - i
            if offset >= 0 and offset <= bar_index
                barHigh = high[offset]
                barLow = low[offset]
                barClose = close[offset]
                barOpen = open[offset]
                barVol = volume[offset]
                
                // Mum bu aralıkta mı kontrol et
                if barHigh >= lowPrice and barLow <= highPrice
                    // Mumun kapsadığı fiyat aralığı
                    candleRange = barHigh - barLow
                    if candleRange > 0
                        // Yukarı/aşağı mum belirleme
                        isUpCandle = barClose >= barOpen
                        
                        // Mumun her satırla kesişimini kontrol et
                        for j = 0 to vpRowCount - 1
                            rowLow = lowPrice + (j * rowHeight)
                            rowHigh = rowLow + rowHeight
                            
                            // Satır ile mum kesişimi
                            overlapLow = math.max(barLow, rowLow)
                            overlapHigh = math.min(barHigh, rowHigh)
                            
                            if overlapHigh > overlapLow
                                // Kesişim oranına göre hacmi dağıt
                                overlapRatio = (overlapHigh - overlapLow) / candleRange
                                volContribution = barVol * overlapRatio
                                
                                if isUpCandle
                                    array.set(upVolumes, j, array.get(upVolumes, j) + volContribution)
                                else
                                    array.set(downVolumes, j, array.get(downVolumes, j) + volContribution)
                                
                                array.set(totalVolumes, j, array.get(totalVolumes, j) + volContribution)
        
        // Maksimum hacmi bul (POC için)
        float maxVolume = 0.0
        int pocIndex = 0
        float totalVolumeSum = 0.0
        
        for i = 0 to vpRowCount - 1
            vol = array.get(totalVolumes, i)
            totalVolumeSum += vol
            if vol > maxVolume
                maxVolume := vol
                pocIndex := i
        
        // Değer alanını hesapla (Value Area)
        float vaVolumeTarget = totalVolumeSum * (vpValueAreaPercent / 100.0)
        float vaVolume = array.get(totalVolumes, pocIndex)
        int vaHighIdx = pocIndex
        int vaLowIdx = pocIndex
        
        while vaVolume < vaVolumeTarget and (vaHighIdx < vpRowCount - 1 or vaLowIdx > 0)
            float upVol = vaHighIdx < vpRowCount - 1 ? array.get(totalVolumes, vaHighIdx + 1) : 0.0
            float dnVol = vaLowIdx > 0 ? array.get(totalVolumes, vaLowIdx - 1) : 0.0
            
            if upVol >= dnVol and vaHighIdx < vpRowCount - 1
                vaHighIdx += 1
                vaVolume += upVol
            else if vaLowIdx > 0
                vaLowIdx -= 1
                vaVolume += dnVol
            else if vaHighIdx < vpRowCount - 1
                vaHighIdx += 1
                vaVolume += upVol
            else
                break
        
        // Kutu genişliğini hesapla
        boxWidth = math.round((endBar - startBar) * (vpWidthPercent / 100.0))
        if boxWidth < 1
            boxWidth := 1
        
        // Hacim profilini çiz
        if maxVolume > 0
            for i = 0 to vpRowCount - 1
                rowLow = lowPrice + (i * rowHeight)
                rowHigh = rowLow + rowHeight
                
                upVol = array.get(upVolumes, i)
                downVol = array.get(downVolumes, i)
                totalVol = array.get(totalVolumes, i)
                
                if totalVol > 0
                    // Normalize edilmiş genişlik
                    volRatio = totalVol / maxVolume
                    barWidth = math.round(boxWidth * volRatio)
                    if barWidth < 1
                        barWidth := 1
                    
                    // Değer alanı içinde mi?
                    inValueArea = i >= vaLowIdx and i <= vaHighIdx
                    
                    if vpVolumeType == "Yukarı/Aşağı"
                        // Yukarı hacim kutusu
                        if upVol > 0
                            upWidth = math.round(barWidth * (upVol / totalVol))
                            if upWidth > 0
                                upBox = box.new(startBar, rowHigh, startBar + upWidth, rowLow,
                                     border_color = color.new(vpUpColor, 100),
                                     bgcolor = inValueArea ? color.new(vpUpColor, vpValueAreasef) : vpUpColor)
                                array.push(vpBoxes, upBox)
                        
                        // Aşağı hacim kutusu
                        if downVol > 0
                            upWidth = math.round(barWidth * (upVol / totalVol))
                            downWidth = barWidth - upWidth
                            if downWidth > 0
                                downBox = box.new(startBar + upWidth, rowHigh, startBar + upWidth + downWidth, rowLow,
                                     border_color = color.new(vpDownColor, 100),
                                     bgcolor = inValueArea ? color.new(vpDownColor, vpValueAreasef) : vpDownColor)
                                array.push(vpBoxes, downBox)
                    
                    else if vpVolumeType == "Toplam"
                        totalBox = box.new(startBar, rowHigh, startBar + barWidth, rowLow,
                             border_color = color.new(vpTotalColor, 100),
                             bgcolor = inValueArea ? color.new(vpTotalColor, vpValueAreasef) : vpTotalColor)
                        array.push(vpBoxes, totalBox)
                    
                    else if vpVolumeType == "Delta"
                        deltaVal = upVol - downVol
                        deltaColor = deltaVal >= 0 ? vpDeltaUpColor : vpDeltaDownColor
                        deltaBox = box.new(startBar, rowHigh, startBar + barWidth, rowLow,
                             border_color = color.new(deltaColor, 100),
                             bgcolor = inValueArea ? color.new(deltaColor, vpValueAreasef) : deltaColor)
                        array.push(vpBoxes, deltaBox)
        
        // POC çizgisi
        if showPocLine
            pocPrice = lowPrice + (pocIndex * rowHeight) + (rowHeight / 2)
            resultPocLine := line.new(startBar, pocPrice, bar_index, pocPrice,
                 color = vpPocColor, width = 2, style = line.style_solid)
        
        // Value Area çizgileri
        if showValueAreaLines
            vahPrice = lowPrice + ((vaHighIdx + 1) * rowHeight)
            valPrice = lowPrice + (vaLowIdx * rowHeight)
            
            resultVahLine := line.new(startBar, vahPrice, bar_index, vahPrice,
                 color = vpValueAreaColor, width = 1, style = line.style_solid)
            resultValLine := line.new(startBar, valPrice, bar_index, valPrice,
                 color = vpValueAreaColor, width = 1, style = line.style_solid)
    
    [resultPocLine, resultVahLine, resultValLine]
